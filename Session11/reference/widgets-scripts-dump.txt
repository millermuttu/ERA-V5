  var m=0,v=0,w=W0,out=[],t,g,mh,vh,st;
  for(t=1;t<=N;t++){
    g=gs[t-1];
    m=B1*m+(1-B1)*g;
    v=B2*v+(1-B2)*g*g;
    mh=bc?m/(1-Math.pow(B1,t)):m;
    vh=bc?v/(1-Math.pow(B2,t)):v;
    st=-ETA*mh/(Math.sqrt(vh)+EPS);
    w+=st;
    out.push({t:t,g:g,m:m,v:v,mh:mh,vh:vh,st:st,w:w,r:st/ETA});
