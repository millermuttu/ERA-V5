    ];
    const dl = Object.fromEntries(dlDefs.map(d => [d[0], d[4]]));
    const dle = id => document.getElementById(id);
    function dlFmt(n){ return n >= 1000000 ? (n/1000000).toFixed(2) + "M" : Math.round(n).toLocaleString(); }
    function dlRenderControls() {
      dle("dlControls").innerHTML = dlDefs.map(d => "<div class='dl-control'><div class='dl-label'><strong>" + d[1] + "</strong><span id='dlv_" + d[0] + "'></span></div><input id='dli_" + d[0] + "' type='range' min='" + d[2] + "' max='" + d[3] + "' value='" + dl[d[0]] + "'></div>").join("");
      dlDefs.forEach(d => { const inp = dle("dli_" + d[0]); inp.addEventListener("input", () => { dl[d[0]] = Number(inp.value); dlRender(); }); });
    }
    function dlCalc() {
      const tokensPerStep = dl.seq * dl.batch;
      const usefulAfterPack = tokensPerStep * dl.pack / 100;
      const usefulAfterOpus = usefulAfterPack * (1 - dl.reject / 100);
      const readMB = tokensPerStep * 2 / (1024*1024) * (1 + dl.compression/100);
      const workerFactor = Math.min(1.8, .35 + Math.log2(dl.workers + 1) / 4 + dl.prefetch / 30);
      const shardPenalty = dl.shard < 256 ? .72 : dl.shard > 3072 ? .88 : 1;
      const effectiveBandwidth = dl.bandwidth * workerFactor * shardPenalty;
      const loaderMs = readMB / effectiveBandwidth * 1000;
      const computeMs = 145 + dl.seq / 8192 * 210 + dl.batch / 1024 * 90;
      const waitMs = Math.max(0, loaderMs - computeMs * .55);
      const stepMs = computeMs + waitMs;
      const idle = waitMs / stepMs * 100;
      return { tokensPerStep, usefulAfterPack, usefulAfterOpus, rejected: usefulAfterPack - usefulAfterOpus, padding: tokensPerStep - usefulAfterPack, loaderMs, computeMs, waitMs, stepMs, idle, usefulTps: usefulAfterOpus / (stepMs/1000), readMB, effectiveBandwidth };
    }
    function dlRender() {
      dlDefs.forEach(d => dle("dlv_" + d[0]).textContent = dl[d[0]] + d[5]);
      const c = dlCalc();
      dle("dlUseful").textContent = dlFmt(c.usefulTps);
      dle("dlIdle").textContent = c.idle.toFixed(0) + "%";
      dle("dlWait").textContent = c.waitMs.toFixed(0) + " ms";
      const total = c.tokensPerStep + c.idle/100*c.tokensPerStep;
      const pct = v => Math.max(2, v / total * 100);
      dle("dlChart").innerHTML = [
        ["Current", c.usefulAfterOpus, c.rejected, c.padding, c.idle/100*c.tokensPerStep],

        ["Tighter packing", c.tokensPerStep*.96*(1-dl.reject/100), c.tokensPerStep*.96*(dl.reject/100), c.tokensPerStep*.04, c.idle/100*c.tokensPerStep],
        ["Fast storage", c.usefulAfterOpus, c.rejected, c.padding, Math.max(0,c.idle-18)/100*c.tokensPerStep]
      ].map(col => "<div class='dl-col'><div class='dl-stack'><div class='dl-seg useful' style='height:" + pct(col[1]) + "%'>use</div><div class='dl-seg rejected' style='height:" + pct(col[2]) + "%'></div><div class='dl-seg padding' style='height:" + pct(col[3]) + "%'></div><div class='dl-seg idle' style='height:" + pct(col[4]) + "%'></div></div><span>" + col[0] + "</span></div>").join("");
      const rows = [
        ["packing", dl.pack, "%", "var(--dl-green)"],
        ["OPUS kept", 100-dl.reject, "%", "var(--dl-violet)"],
        ["bandwidth", Math.min(100, c.effectiveBandwidth/4500*100), "%", "var(--dl-cyan)"],
        ["worker cover", Math.min(100, (dl.workers/32*70 + dl.prefetch/16*30)), "%", "var(--dl-blue)"],
        ["GPU busy", 100-c.idle, "%", c.idle > 20 ? "var(--dl-amber)" : "var(--dl-green)"]
      ];
      dle("dlBars").innerHTML = rows.map(r => "<div class='dl-row'><span>" + r[0] + "</span><div class='dl-bar'><div class='dl-fill' style='width:" + Math.max(2,r[1]).toFixed(0) + "%;background:" + r[3] + "'></div></div><span>" + r[1].toFixed(0) + r[2] + "</span></div>").join("");
      let advice = "Healthy pipeline: most step time becomes useful training tokens.";
      if (dl.pack < 75) advice = "Packing is the first bottleneck. More workers will not fix empty context windows.";
      else if (dl.reject > 45) advice = "OPUS is discarding too much prepared data. Track accepted tokens/sec, not just raw tokens/sec.";
      else if (c.idle > 20) advice = "The GPU is waiting. Increase cache/prefetch, reduce tiny shards, or improve storage bandwidth.";
      else if (dl.shard < 256) advice = "Small shards create metadata and open-file overhead. Bundle into larger immutable objects.";
      dle("dlAdvice").textContent = advice;
    }
