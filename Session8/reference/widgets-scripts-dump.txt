    for(i=0;i<N;i++) for(d=0;d<D;d++) H[i*D+d]=Hs[i][d];
    for(i=0;i<D*NV;i++) Wout[i]=gau()*0.05;
    var gW=new Float64Array(D*NV), gb=new Float64Array(NV), lg=new Float64Array(NV);
    var STEPS=300, LR=1.1, DECAY=0.012;
    for(var step=0;step<STEPS;step++){
      gW.fill(0); gb.fill(0);
      for(i=0;i<N;i++){
        var base=i*D, m=-Infinity;
        for(v=0;v<NV;v++){ var z=bout[v]; for(d=0;d<D;d++) z+=H[base+d]*Wout[d*NV+v]; lg[v]=z; if(z>m) m=z; }
        s2=0; for(v=0;v<NV;v++){ lg[v]=Math.exp(lg[v]-m); s2+=lg[v]; }
        for(v=0;v<NV;v++) lg[v]/=s2;
        lg[Ys[i]]-=1;
        for(d=0;d<D;d++){ var hv=H[base+d]; if(hv===0) continue; var o2=d*NV;
          for(v=0;v<NV;v++) gW[o2+v]+=hv*lg[v]; }
        for(v=0;v<NV;v++) gb[v]+=lg[v];
      }
      var inv=1/N;
      for(i=0;i<D*NV;i++) Wout[i]-=LR*(gW[i]*inv+DECAY*Wout[i]);
      for(v=0;v<NV;v++) bout[v]-=LR*gb[v]*inv;
    }
  })();

  function logitsOf(fin){ var lg=new Float64Array(NV);
    for(var v=0;v<NV;v++){ var z=bout[v]; for(var d=0;d<D;d++) z+=fin[d]*Wout[d*NV+v]; lg[v]=z; } return lg; }

  function distribution(lg,T,K){
    var order=[]; for(var i=0;i<NV;i++) order.push(i);
    order.sort(function(a,b){ return lg[b]-lg[a]; });
    var keep=order.slice(0,K), m=-Infinity, i2;
    for(i2=0;i2<keep.length;i2++){ var z=lg[keep[i2]]/T; if(z>m) m=z; }
    var ex=[], s=0;
    for(i2=0;i2<keep.length;i2++){ var e=Math.exp(lg[keep[i2]]/T-m); ex.push(e); s+=e; }
    var p=new Float64Array(NV);
    for(i2=0;i2<keep.length;i2++) p[keep[i2]]=ex[i2]/s;
    return {p:p, order:order};
  }

  function fl(x){return x>=1e15?(x/1e15).toFixed(2)+' PFLOP':x>=1e12?(x/1e12).toFixed(2)+' TFLOP':x>=1e9?(x/1e9).toFixed(2)+' GFLOP':(x/1e6).toFixed(1)+' MFLOP';}
  function num(n){return Math.round(n).toLocaleString('en-US');}

  function state(){
    var L=+el.L.value,D=+el.D.value,sch=se.sched.value;
    var share=(sch==='custom')?(+el.R.value)/100:SHARE[sch];
    var La=Math.round(L*share), Ll=L-La;
    var m=+el.M.value,k=+el.K.value,w=+el.W.value,kind=se.kind.value;
    var eff_m=(kind==='gqa')?1:(kind==='hca'?m*4:m);
    var T=ctx();
    var blocks=Math.ceil(T/eff_m);
    var scored=(kind==='gqa')?T/2:(kind==='hca'?blocks+w:Math.min(k,blocks)+w);
    var held=(kind==='gqa')?T:blocks+w;
    return {L:L,D:D,sch:sch,share:share,La:La,Ll:Ll,m:m,k:k,w:w,kind:kind,eff_m:eff_m,
      T:T,blocks:blocks,scored:scored,held:held,
      Q:+el.Q.value,G:+el.G.value,Hd:+el.H.value,prec:parseFloat(se.prec.value),
      base:Math.pow(10,+el.B.value),rot:+el.Rd.value,S:pow2(+el.S.value),
      pos:se.pos.value,extp:se.ext.value,mem:se.mem.value,chunk:pow2(+el.C.value),
      tok:(+el.Kt.value)*1e9,budget:+el.Bu.value,want:+el.Cc.value,dev:+el.N.value,
      N:12*L*D*D};
  }
  function cache(s,T){
    var blocks=(s.kind==='gqa')?0:Math.ceil(T/s.eff_m);
    var held=(s.kind==='gqa')?T:blocks+s.w;
    return 2*s.La*s.G*s.Hd*held*s.prec + s.Ll*s.D*s.Hd*s.prec;
  }
  function attnFlops(s,T){
    var blocks=(s.kind==='gqa')?0:Math.ceil(T/s.eff_m);
    var scored=(s.kind==='gqa')?T/2:(s.kind==='hca'?blocks+s.w:Math.min(s.k,blocks)+s.w);
    var qd=s.Q*s.Hd;
    return 4*s.La*qd*scored + (s.kind==='csa'?4*s.La*IDX_DIM*blocks:0) + 4*s.Ll*s.D*s.Hd;
  }
  function fwdPerToken(s,T){return 2*s.N+attnFlops(s,T);}
  function trainDays(s){return 3*fwdPerToken(s,s.S)*s.tok/(PEAK*MFU*86400);}
  function reach(s){return s.pos==='learned'?s.S:(s.pos==='alibi'?Infinity:s.S*EXT[s.extp]);}
  function bandsAlive(s,T){if(s.rot<2)return 0;var a=0,n=s.rot/2;for(var j=0;j<n;j++){if(2*Math.PI*Math.pow(s.base,2*j/s.rot)>=T)a++;}return a;}
  function minBase(s,T){return s.rot>2?Math.pow(T/(2*Math.PI),s.rot/(s.rot-2)):Infinity;}
  function gapLayers(s){return s.La>0?Math.ceil(s.L/s.La)-1:s.L;}

'+

Your exact-attention share is <code>'+(s.share*100).toFixed(0)+'%</code>, putting <code>'+gp+'</code> fixed-state layers between exact layers at <code>'+gb(cb)+'</code> of cache. A different ratio moves both numbers by an unmeasured amount.',

    if(s.Ll>0&&s.kind!=='gqa'){
      oq.push({id:'linear_vs_compression',bears:'both levers active',
        q:'Linear state or sequence compression.',
        b:'You are using both. V4 and Qwen3.6 chose the linear state, DeepSeek chose compression, and no published result compares them at matched budget.         b:'You chose the linear state, as V4 and Qwen3.6 did. DeepSeek chose compression and reports the stronger cache number. The two have never been compared at matched budget.',

  }

  function renderRope(j,i){
    var base=25*Math.PI/180;
    var ka=state.rope?base+j*THETA:base;
    var qa=state.rope?base+i*THETA:base;
    var rel=state.rope?(i-j)*THETA:0;

  el('shift').addEventListener('input',function(){state.shift=Number(this.value);render();});
  el('ropeToggle').addEventListener('click',function(){state.rope=!state.rope;render();});

  });

  render();
})();

    if(n>=1e12)return (n/1e12).toFixed(2)+' TB';

    if(n>=1e6)return (n/1e6).toFixed(1)+' MB';
    if(n>=1e3)return (n/1e3).toFixed(1)+' KB';
