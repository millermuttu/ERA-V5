function build(p,m,sch){
  var g=[],tF=[],tB=[],T=0,i,j;
  for(i=0;i<p;i++){g.push([]);tF.push([]);tB.push([]);}
  function put(i,t,k,j){g[i][t]={k:k,j:j};if(t+1>T)T=t+1;}
  if(sch==='afab'){
    for(i=0;i<p;i++){for(j=0;j<m;j++){put(i,i+j,'F',j);put(i,(m+p-1)+j+(p-1-i),'B',j);}}
    return {g:g,T:T};
  }
  /* Non-interleaved 1F1B: stage i runs p-1-i warm-up forwards, then alternates F and B, then drains. */
  var ops=[],pos=[],free=[],left=0,w,r,o,op,rd,t,prog,src;
  for(i=0;i<p;i++){
    w=Math.min(p-1-i,m);o=[];
    for(j=0;j<w;j++)o.push(['F',j]);
    for(r=0;r<m-w;r++){o.push(['F',w+r]);o.push(['B',r]);}
    for(j=m-w;j<m;j++)o.push(['B',j]);
    ops.push(o);pos.push(0);free.push(0);left+=o.length;
  }
  while(left>0){
    prog=false;
    for(i=0;i<p;i++){
      if(pos[i]>=ops[i].length)continue;
      op=ops[i][pos[i]];
      if(op[0]==='F'){src=i===0?-1:tF[i-1][op[1]];rd=i===0?0:(src===undefined?null:src+1);}
      else{src=i===p-1?tF[i][op[1]]:tB[i+1][op[1]];rd=src===undefined?null:src+1;}
      if(rd===null)continue;
      t=Math.max(rd,free[i]);
      (op[0]==='F'?tF:tB)[i][op[1]]=t;put(i,t,op[0],op[1]);
      free[i]=t+1;pos[i]++;left--;prog=true;
    }
    if(!prog)break;
  }
  return {g:g,T:T};
}
