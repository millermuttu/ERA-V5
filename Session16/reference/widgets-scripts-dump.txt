function pct(arr,p){var s=arr.slice().sort(function(x,y){return x-y;});return s[Math.min(s.length-1,Math.floor(p*(s.length-1)+0.5))];}
function px(lx){return L+(lx-X0)/(X1-X0)*(W-L-R);}
function py(v){v=Math.max(Y0,Math.min(Y1,v));return T+(Y1-v)/(Y1-Y0)*(H-T-B);}
function draw(){
  var zr=rng(1000+n),rungs=[],i,j,k,rep=src?REP:1,train=[],held,f,boots=[],br=rng(77+n*13+hold),s='',lx,
      xs=[],lo=[],hi=[],grid=60,hi24,lo24,p24,ph,hl,hh,inside,real24,real24in;
  for(i=0;i<n;i++){var lc=LO+(HI-LO)*i/(n-1),z=gauss(zr);
    rungs.push({lc:lc,b:E+(truth(lc,rep)-E)*Math.exp(sig*z)});}
  held=rungs[n-hold];
  for(i=0;i<n;i++){if(i!==n-hold)train.push(rungs[i]);}
  f=fit(train);
  for(k=0;k<NB;k++){var smp=[],seen={},cnt=0;
    for(j=0;j<train.length;j++){var r=train[Math.floor(br()*train.length)];smp.push(r);if(!seen[r.lc]){seen[r.lc]=1;cnt++;}}
    if(cnt<2)continue;var bf=fit(smp);if(bf)boots.push(bf);}
  for(k=0;k<=grid;k++){lx=X0+0.5+(X1-X0-0.5)*k/grid;xs.push(lx);
    var v=boots.map(function(b){return pred(b,lx);});lo.push(pct(v,0.05)-TOL);hi.push(pct(v,0.95)+TOL);}
  var bh=boots.map(function(b){return pred(b,held.lc);}),b24=boots.map(function(b){return pred(b,TGT);});
  hl=pct(bh,0.05)-TOL;hh=pct(bh,0.95)+TOL;ph=pred(f,held.lc);inside=held.b>=hl&&held.b<=hh;
  lo24=pct(b24,0.05)-TOL;hi24=pct(b24,0.95)+TOL;p24=pred(f,TGT);
  real24=truth(TGT,REP);real24in=real24>=lo24&&real24<=hi24;
  // axes
  for(i=0;i<=6;i++){var yv=Y0+(Y1-Y0)*i/6,y=py(yv);
    s+='<line x1="'+L+'" x2="'+(W-R)+'" y1="'+y+'" y2="'+y+'" stroke="#e4ebf4"/><text x="'+(L-6)+'" y="'+(y+3.5)+'" text-anchor="end" font-size="10" font-family="ui-monospace,monospace" fill="#93a3b6">'+yv.toFixed(1)+'</text>';}
  for(i=19;i<=24;i++){s+='<text x="'+px(i)+'" y="'+(H-B+15)+'" text-anchor="middle" font-size="10" font-family="ui-monospace,monospace" fill="#93a3b6">1e'+i+'</text>';}
  s+='<text x="'+((L+W-R)/2)+'" y="'+(H-4)+'" text-anchor="middle" font-size="10.5" font-weight="700" fill="#65758b">training compute C in FLOPs</text>';
  s+='<text x="12" y="'+((T+H-B)/2)+'" text-anchor="middle" font-size="10.5" font-weight="700" fill="#65758b" transform="rotate(-90 12 '+((T+H-B)/2)+')">held-out BPB</text>';
  s+='<rect x="'+px(LO-0.15)+'" y="'+T+'" width="'+(px(HI+0.15)-px(LO-0.15))+'" height="'+(H-T-B)+'" fill="#f4f7fb"/>';
  s+='<text x="'+px((LO+HI)/2)+'" y="'+(T+14)+'" text-anchor="middle" font-size="10.5" font-weight="700" fill="#93a3b6">ladder, 1e19 to 1e22</text>';
  s+='<text x="'+px(TGT)+'" y="'+(T+14)+'" text-anchor="middle" font-size="10.5" font-weight="700" fill="#93a3b6">large run</text>';
  // band
  var up='',dn='';
  for(k=0;k<xs.length;k++){up+=(k?'L':'M')+px(xs[k]).toFixed(1)+' '+py(hi[k]).toFixed(1)+' ';dn=' L'+px(xs[k]).toFixed(1)+' '+py(lo[k]).toFixed(1)+dn;}
  s+='<path d="'+up+dn+' Z" fill="#7c3aed" opacity=".16"/>';
  var fl='';for(k=0;k<xs.length;k++){fl+=(k?'L':'M')+px(xs[k]).toFixed(1)+' '+py(pred(f,xs[k])).toFixed(1)+' ';}
  s+='<path d="'+fl+'" fill="none" stroke="'+VI+'" stroke-width="2.2"/>';
  // forecast bar
  s+='<line x1="'+px(TGT)+'" x2="'+px(TGT)+'" y1="'+py(hi24)+'" y2="'+py(lo24)+'" stroke="'+VI+'" stroke-width="3"/>';
  s+='<circle cx="'+px(TGT)+'" cy="'+py(p24)+'" r="4.5" fill="'+VI+'"/>';
  s+='<circle cx="'+(px(TGT)+16)+'" cy="'+py(real24)+'" r="6" fill="#fff" stroke="'+AM+'" stroke-width="2.6"/>';
  for(i=0;i<train.length;i++){s+='<circle cx="'+px(train[i].lc)+'" cy="'+py(train[i].b)+'" r="5.5" fill="'+CY+'"/>';}
  s+='<circle cx="'+px(held.lc)+'" cy="'+py(held.b)+'" r="7" fill="#fff" stroke="'+(inside?GR:RD)+'" stroke-width="2.8"/>';
  $('g').innerHTML=s;
  $('h2').textContent='held-out BPB against training FLOPs, '+n+' rungs, noise '+Math.round(sig*100)+'%, source scaling '+(src?'on':'off');
  $('r1').innerHTML='<b style="color:'+VI+'">BPB = 0.60 + '+f.a.toFixed(3)+' &middot; (C / 1e19)^&minus;'+f.al.toFixed(3)+'</b>';
  $('r1b').innerHTML='fit on '+train.length+' rungs, '+boots.length+' bootstrap resamples';
  $('l2').textContent='held-out rung at '+sci(Math.pow(10,held.lc))+' FLOPs, '+(inside?'inside the band':'outside the band');
  $('r2').innerHTML='<b style="color:'+(inside?GR:RD)+'">measured '+held.b.toFixed(3)+'</b>, predicted '+ph.toFixed(3);
  $('r2b').innerHTML='band '+hl.toFixed(3)+' to '+hh.toFixed(3);
  $('r3').innerHTML='<b style="color:'+VI+'">'+p24.toFixed(3)+'</b> = 0.60 + '+f.a.toFixed(3)+' &middot; (1e5)^&minus;'+f.al.toFixed(3);
  $('r3b').innerHTML='band '+lo24.toFixed(3)+' to '+hi24.toFixed(3)+', large run result <b style="color:'+AM+'">'+real24.toFixed(3)+'</b>';
  var msg;
  if(!inside){
    msg='<span style="color:'+RD+'">The held-out rung is outside the band. Do not trust this forecast. Add rungs or reduce the noise, then test again.</span>';
  }else if(!src){
    msg='<span style="color:'+RD+'">The held-out rung is inside the band. The large run result is '+real24.toFixed(3)+(real24in?', near the top of the forecast band.':', above the forecast band.')+' Without source scaling, the small rungs see no repeated data. The large run repeats its sources. Thus the forecast is too low.</span>';
  }else if(real24in){
    msg='<span style="color:'+GR+'">The held-out rung is inside the band. Each rung repeats each source as often as the large run does. The large run result lands inside the forecast band.</span>';
  }else{
    msg='<span style="color:'+AM+'">The held-out rung is inside the band. The large run result is '+real24.toFixed(3)+', outside the forecast band. The forecast reaches two orders of compute past the top rung. Use a wider band or add a higher rung.</span>';
  }
  $('note').innerHTML=msg;
}

  });
}

    b.addEventListener('click',function(){set(+b.getAttribute('data-v'));marks();draw();});
  });
}
